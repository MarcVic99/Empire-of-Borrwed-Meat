const int size = 96, padding = 10, cell = size + padding * 2, frames = 12;
const string root = "Assets/Frankenstein/Resources/Frankenstein/Imperio/";
const string clips = "Assets/Frankenstein/Art/Imperio/Animations/";
var roles = new[] { Frankenstein.Profession.Hunter, Frankenstein.Profession.Dancer, Frankenstein.Profession.Grafter };
var directions = new[] { "south", "west", "east", "north" };
foreach (var role in roles)
{
    string id = Frankenstein.Presentation.ImperialArt.Role(role);
    var original = new UnityEngine.Texture2D(2, 2);
    UnityEngine.ImageConversion.LoadImage(original, System.IO.File.ReadAllBytes(root + "walk-" + id + ".png"));
    var atlas = new UnityEngine.Texture2D(cell * frames, cell * 4, UnityEngine.TextureFormat.RGBA32, false);
    var pixels = new UnityEngine.Color32[cell * frames * cell * 4];
    const string dancerSheet = "Assets/Frankenstein/ArtSource/Imperio/dancer-walk-keyframes.png";
    UnityEngine.Texture2D drawn = null;
    if (role == Frankenstein.Profession.Dancer && System.IO.File.Exists(dancerSheet))
    {
        drawn = new UnityEngine.Texture2D(2, 2);
        UnityEngine.ImageConversion.LoadImage(drawn, System.IO.File.ReadAllBytes(dancerSheet));
    }
    for (int row = 0; row < 4; row++)
    {
        if (drawn != null && row != 0)
        {
            int sourceWidth = drawn.width / 4;
            int sourceHeight = drawn.height / 2;
            int baseY = row == 3 ? 0 : sourceHeight;
            var bounds = new UnityEngine.RectInt[4];
            int tallest = 0;
            for (int key = 0; key < 4; key++)
            {
                int minX = sourceWidth, maxX = -1, minY = sourceHeight, maxY = -1;
                for (int y = 0; y < sourceHeight; y++)
                for (int x = 0; x < sourceWidth; x++)
                {
                    if (drawn.GetPixel(key * sourceWidth + x, baseY + y).a < .25f) continue;
                    minX = System.Math.Min(minX, x); maxX = System.Math.Max(maxX, x);
                    minY = System.Math.Min(minY, y); maxY = System.Math.Max(maxY, y);
                }
                if (maxX < minX || maxY < minY || minX == 0 || maxX == sourceWidth - 1 ||
                    minY == 0 || maxY == sourceHeight - 1)
                    throw new System.Exception("Drawn gait needs clear gutters");
                bounds[key] = new UnityEngine.RectInt(minX, minY, maxX - minX + 1, maxY - minY + 1);
                tallest = System.Math.Max(tallest, bounds[key].height);
            }

            float scale = 84f / tallest;
            for (int frame = 0; frame < frames; frame++)
            {
                int phase = frame / 3;
                // Mirror the two rear poses for the other half of the stride.
                int key = row == 3 ? (phase % 2 == 0 ? 0 : 3) : phase;
                bool mirror = row == 2 || row == 3 && phase >= 2;
                var sourceBounds = bounds[key];
                int width = UnityEngine.Mathf.RoundToInt(sourceBounds.width * scale);
                int height = UnityEngine.Mathf.RoundToInt(sourceBounds.height * scale);
                int center = cell / 2;
                int top = padding + 5;
                for (int y = 0; y < height; y++)
                for (int x = 0; x < width; x++)
                {
                    int sourceX = sourceBounds.x + UnityEngine.Mathf.FloorToInt((mirror ? width - x - .5f : x + .5f) * sourceBounds.width / width);
                    int sourceY = sourceBounds.yMax - 1 - UnityEngine.Mathf.FloorToInt((y + .5f) * sourceBounds.height / height);
                    int dst = ((4 - row) * cell - top - y - 1) * cell * frames + frame * cell + center - width / 2 + x;
                    var color = drawn.GetPixel(key * sourceWidth + sourceX, baseY + sourceY);
                    if (color.a >= .25f) pixels[dst] = color;
                }
            }
            continue;
        }
        var uv = Frankenstein.Presentation.ExplorationArt.FrameUV(role, row, 0);
        var source = new UnityEngine.Color32[size * size];
        int footLeft = size, footRight = 0, soleY = 0;
        for (int y = 0; y < size; y++)
        for (int x = 0; x < size; x++)
        {
            var color = original.GetPixel(
                UnityEngine.Mathf.FloorToInt((uv.x + (x + .5f) / size * uv.width) * original.width),
                UnityEngine.Mathf.FloorToInt((uv.y + (size - y - .5f) / size * uv.height) * original.height));
            source[y * size + x] = color;
            if (y > 82 && color.a > .5f) { footLeft = System.Math.Min(x, footLeft); footRight = System.Math.Max(x, footRight); soleY = System.Math.Max(y, soleY); }
        }
        bool profile = row == 1 || row == 2;
        float seam = footLeft < footRight ? (footLeft + footRight) * .5f : size * .5f;
        for (int frame = 0; frame < frames; frame++)
        {
            float phase = frame / (float)frames;
            // A foot moves backward at a steady rate during support, then swings forward.
            float contact = UnityEngine.Mathf.Repeat(phase * 2, 1);
            int bob = row == 3 ? 0 : UnityEngine.Mathf.RoundToInt(UnityEngine.Mathf.Sin(contact * UnityEngine.Mathf.PI));
            float transfer = UnityEngine.Mathf.Cos((phase - .08f) * 2 * UnityEngine.Mathf.PI);
            float lean = UnityEngine.Mathf.Sin(contact * UnityEngine.Mathf.PI);
            var pose = new UnityEngine.Color32[cell * cell];
            for (int pass = 0; pass < 2; pass++)
            {
                bool right = row == 1 ? pass == 0 : pass == 1;
                float step = UnityEngine.Mathf.Repeat(phase + (right ? .5f : 0), 1);
                const float stance = .58f;
                bool supporting = step < stance;
                float swing = UnityEngine.Mathf.Clamp01((step - stance) / (1 - stance));
                float travel = supporting ? UnityEngine.Mathf.Lerp(1, -1, step / stance)
                    : UnityEngine.Mathf.SmoothStep(-1, 1, swing);
                float lift = supporting ? 0 : UnityEngine.Mathf.Sin(swing * UnityEngine.Mathf.PI);
                float armSwing = -UnityEngine.Mathf.Cos(step * 2 * UnityEngine.Mathf.PI);
                float forward = row == 0 || row == 2 ? 1 : -1;
                for (int y = -padding; y < size + padding; y++)
                {
                    float leg = UnityEngine.Mathf.SmoothStep(0, 1, (y - 55f) / 32f);
                    float upper = 1 - leg;
                    float armHeight = UnityEngine.Mathf.SmoothStep(0, 1, (y - 37f) / 17f)
                        * (1 - UnityEngine.Mathf.SmoothStep(0, 1, (y - 60f) / 11f));
                    for (int x = -padding; x < size + padding; x++)
                    {
                        // Shoulder stays attached while the visible forearm and hand swing.
                        float arm = armHeight * UnityEngine.Mathf.SmoothStep(0, 1,
                            (UnityEngine.Mathf.Abs(x - seam) - (profile ? 4 : 10)) / 8f);
                        float strideY = profile ? 0 : travel * forward * (row == 3 ? 6 : 4);
                        float liftY = lift * (profile ? 4 : row == 3 ? 2 : 3);
                        float shoulder = 1 - UnityEngine.Mathf.SmoothStep(0, 1, (y - 42f) / 29f);
                        float dx = profile ? (travel * 6 * leg + armSwing * 3 * arm + lean * shoulder) * forward
                            : -transfer * 1.7f * upper;
                        if (row == 3 && role != Frankenstein.Profession.Hunter)
                            dx += (right ? 1 : -1) * 2.4f * UnityEngine.Mathf.SmoothStep(0, 1, (y - 70f) / 20f);
                        float dy = (strideY - liftY) * leg - bob * upper;
                        if (!profile) dy += armSwing * forward * 3 * arm;
                        int sx = x - UnityEngine.Mathf.RoundToInt(dx);
                        int sy = y - UnityEngine.Mathf.RoundToInt(dy);
                        if (sy < 0 || sy >= size || sx < 0 || sx >= size) continue;
                        // Keep the near toe and heel together below the far shoe.
                        float split = seam;
                        if (profile)
                        {
                            float sole = UnityEngine.Mathf.SmoothStep(0, 1, (sy - (soleY - 10f)) / 7f);
                            split = UnityEngine.Mathf.Lerp(seam, row == 1 ? footRight + 1 : footLeft - 1, sole);
                        }
                        if ((sx >= split) != right) continue;
                        var color = source[sy * size + sx];
                        if (color.a < 48) continue;
                        // The far foot reads as the far foot even at map scale.
                        if (profile && (right == (row == 1)) && sy > 77)
                        {
                            color.r = (byte)(color.r * .83f);
                            color.g = (byte)(color.g * .83f);
                            color.b = (byte)(color.b * .83f);
                        }
                        pose[(y + padding) * cell + x + padding] = color;
                    }
                }
            }
            for (int y = 0; y < cell; y++)
            for (int x = 0; x < cell; x++)
                pixels[((4 - row) * cell - y - 1) * cell * frames + frame * cell + x] = pose[y * cell + x];
        }
    }
    if (drawn != null) UnityEngine.Object.DestroyImmediate(drawn);
    atlas.SetPixels32(pixels); atlas.Apply();
    string path = root + "gait-" + id + ".png";
    System.IO.File.WriteAllBytes(path, UnityEngine.ImageConversion.EncodeToPNG(atlas));
    UnityEngine.Object.DestroyImmediate(atlas); UnityEngine.Object.DestroyImmediate(original);
    UnityEditor.AssetDatabase.ImportAsset(path, UnityEditor.ImportAssetOptions.ForceSynchronousImport);
    var importer = (UnityEditor.TextureImporter)UnityEditor.AssetImporter.GetAtPath(path);
    importer.textureType = UnityEditor.TextureImporterType.Sprite;
    importer.spriteImportMode = UnityEditor.SpriteImportMode.Multiple;
    importer.spritePixelsPerUnit = 64f * size / 314f;
    importer.filterMode = UnityEngine.FilterMode.Point;
    importer.mipmapEnabled = false;
    importer.npotScale = UnityEditor.TextureImporterNPOTScale.None;
    importer.textureCompression = UnityEditor.TextureImporterCompression.Uncompressed;
    var factory = new UnityEditor.U2D.Sprites.SpriteDataProviderFactories(); factory.Init();
    var provider = factory.GetSpriteEditorDataProviderFromObject(importer); provider.InitSpriteEditorDataProvider();
    var prior = provider.GetSpriteRects();
    var records = new UnityEditor.SpriteRect[frames * 4];
    for (int row = 0; row < 4; row++)
    for (int frame = 0; frame < frames; frame++)
    {
        string name = id + "-" + directions[row] + "-" + frame;
        var old = System.Array.Find(prior, sprite => sprite.name == name);
        records[row * frames + frame] = new UnityEditor.SpriteRect {
            name = name, spriteID = old == null ? UnityEngine.GUID.Generate() : old.spriteID,
            rect = new UnityEngine.Rect(frame * cell, (3 - row) * cell, cell, cell),
            pivot = new UnityEngine.Vector2(.5f, padding / (float)cell), alignment = UnityEngine.SpriteAlignment.Custom
        };
    }
    provider.SetSpriteRects(records);
    var ids = provider.GetDataProvider<UnityEditor.U2D.Sprites.ISpriteNameFileIdDataProvider>();
    ids.SetNameFileIdPairs(System.Linq.Enumerable.Select(records, sprite => new UnityEditor.SpriteNameFileIdPair(sprite.name, sprite.spriteID)));
    provider.Apply(); importer.SaveAndReimport();
    var sprites = System.Linq.Enumerable.ToDictionary(System.Linq.Enumerable.OfType<UnityEngine.Sprite>(UnityEditor.AssetDatabase.LoadAllAssetsAtPath(path)), s => s.name);
    for (int row = 0; row < 4; row++)
    {
        var clip = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.AnimationClip>(clips + id + "-walk-" + directions[row] + ".anim");
        clip.frameRate = 16;
        var keys = new UnityEditor.ObjectReferenceKeyframe[frames + 1];
        for (int i = 0; i <= frames; i++) keys[i] = new UnityEditor.ObjectReferenceKeyframe { time = i / 16f, value = sprites[id + "-" + directions[row] + "-" + (i % frames)] };
        var binding = new UnityEditor.EditorCurveBinding { type = typeof(UnityEngine.SpriteRenderer), propertyName = "m_Sprite" };
        UnityEditor.AnimationUtility.SetObjectReferenceCurve(clip, binding, keys);
        UnityEditor.AnimationUtility.SetEditorCurve(clip, new UnityEditor.EditorCurveBinding { type = typeof(UnityEngine.SpriteRenderer), propertyName = "m_FlipX" }, null);
        UnityEditor.AnimationUtility.SetEditorCurve(clip, new UnityEditor.EditorCurveBinding { path = "Head", type = typeof(UnityEngine.GameObject), propertyName = "m_IsActive" }, UnityEngine.AnimationCurve.Constant(0, frames / 16f, 0));
        UnityEditor.EditorUtility.SetDirty(clip);
    }
}
UnityEditor.AssetDatabase.SaveAssets();
return "144 gait poses and 12 walk clips";
